반경 안의 아파트만 찾는 SQL

하버사인(Haversine) 공식 · LIMIT 페이징 · COUNT 쿼리

-- House.xml (MyBatis)
SELECT aptCode
FROM (
  SELECT aptCode,
    (6371 * ACOS(
       COS(RADIANS(#{lat})) * COS(RADIANS(lat)) * COS(RADIANS(lng) - RADIANS(#{lng}))
     + SIN(RADIANS(#{lat})) * SIN(RADIANS(lat))
    )) AS distance  -- km
  FROM houseinfo
) AS subquery
WHERE distance < #{dis}
